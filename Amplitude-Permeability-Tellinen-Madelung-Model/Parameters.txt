# ============================================================
# Tellinen-Madelung amplitude-permeability calculation
# ============================================================

# ------------------------------------------------------------
# FIELD AMPLITUDE SCAN
# ------------------------------------------------------------

# Initial magnetic-field amplitude, A/m.
# Every calculation starts at H = +Hstart on the upper major branch.
Hstart=1000.0

# Final magnetic-field amplitude, A/m.
# Hb is scanned downward from Hstart to Hstop.
Hstop=10.0

# Number of Hb values in the interval [Hstop, Hstart],
# including both endpoints.
HbPoints=30

# ------------------------------------------------------------
# NUMERICAL TOLERANCES
# ------------------------------------------------------------
H_tol=1.0e-9
B_tol=1.0e-9
D_tol=1.0e-12

# ------------------------------------------------------------
# NUMERICAL INTEGRATION
# ------------------------------------------------------------
# Allowed values:
#   trapezoid  - linear formula, global error O(dH^2)
#   lagrange   - 3-point quadratic Lagrange formula,
#                global error O(dH^4) for smooth functions
IntegrationMethod=lagrange

# Number of subintervals used for each definite integral.
# For 'lagrange' an even number is used; if an odd number is
# specified, the program automatically increases it by one.
IntegrationIntervals=500
